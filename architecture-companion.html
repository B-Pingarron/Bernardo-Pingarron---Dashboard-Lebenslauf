// Lines 1-3: Import Node.js built-ins const fs = require('fs'); const path = require('path'); // Lines 4-6: Resolve file paths relative to script location const templatePath = path.join(__dirname, '../src/template.html'); const translationsDir = path.join(__dirname, '../translations'); const outputDir = path.join(__dirname, '../dist'); // Lines 8-9: Create dist/ if missing if (!fs.existsSync(outputDir)) fs.mkdirSync(outputDir, { recursive: true }); // Line 12: Read template ONCE (shared across all languages) const template = fs.readFileSync(templatePath, 'utf8'); // Lines 15-30: Loop through each translation file fs.readdirSync(translationsDir).forEach(file => { if(!file.endsWith('.json')) return; const lang = path.basename(file, '.json'); // "de", "en", "es" const translations = JSON.parse( // Parse JSON fs.readFileSync(path.join(translationsDir, file), 'utf8') ); let html = template; // Replace ALL {{key}} patterns with translated values html = html.replace( new RegExp(`\\{\\{(${key})\\}\\}`, 'g'), // Pattern: {{any_word}} value // Replacement: translated text ); // Special case: {{lang}} → "de" / "en" / "es" html = html.replace('{{lang}}', lang); // Write to dist/dashboard_{lang}.html fs.writeFileSync(outputPath, html, 'utf8'); });

⚠ THE GAP: Chart Labels Not Translated

The build system translates HTML text via {{placeholders}} (70 keys), but the JavaScript Canvas code has hardcoded German labels for radars, timeline roles, and donut segments. These are the same in every language output. The charts render correctly but show German text regardless of which translation is used.

6. The Fix: Data-Driven Approach

● Current: Hardcoded in JS

  • Labels scattered across template.html
  • Same German text in every output
  • ~25 strings to translate manually
  • Mix of translated (HTML) + untranslated (JS)
  • Template readability suffers

● Target: Data-Driven

  • All chart data in one CHART_DATA object
  • Build script injects per-language JSON
  • JS reads from CHART_DATA, not hardcoded
  • Zero new keys needed — labels in data object
  • Template stays clean, data stays separate
// TARGET: At top of page, build script injects this: <script> const CHART_DATA = { radar_prof: ["Anpassung", "Innovation", "Sozial", "Unabhängigkeit", "Leistung"], radar_pers: ["Empathie", "Strategie", "Lernbereitschaft", "Kreativ", "Perspektive", "Emotionale Intelligenz"], radar_soft: ["Problem Lösung", "Teamführung", "Workflow", "Agilität", "Anpassungsfähigkeit"], timeline: [ {label:"Casa Negra", sub:"Barback → Bartender", start:2013.42, end:2014.92, color:"gold", row:0, stemH:38}, // ... 7 more roles ], donut: [ {pct:0.65, color:"#B8922A"}, {pct:0.20, color:"#2D5016"}, {pct:0.15, color:"#7D1A1A"} ] }; </script> // Drawing functions read from CHART_DATA: drawRadar('radarProf', CHART_DATA.radar_prof, [...], ...); drawTimeline(CHART_DATA.timeline);
